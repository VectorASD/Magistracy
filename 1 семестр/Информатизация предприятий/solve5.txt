25.	Сотрудника, у которого EMPLOYEE_ID=200, перевести в подчинение к тому же менеджеру, как у EMPLOYEE_ID=105, и установить зарплату на 10% выше минимума для занимаемой должности.

UPDATE EMPLOYEES
SET MANAGER_ID = (
      SELECT MANAGER_ID
      FROM EMPLOYEES
      WHERE EMPLOYEE_ID = 105
    ), -- 103
    SALARY = (
      SELECT MIN_SALARY * 1.1
      FROM JOBS
      WHERE JOB_ID = (
        SELECT JOB_ID
        FROM EMPLOYEES
        WHERE EMPLOYEE_ID = 200
      ) -- 'AD_ASST'
    ) -- 3300
WHERE EMPLOYEE_ID = 200; -- 1 row updated.

-- Проверка:
SELECT EMPLOYEE_ID, MANAGER_ID, SALARY FROM EMPLOYEES WHERE EMPLOYEE_ID = 200;

34.	На основе таблицы HR.EMPLOYEES создать таблицу EMP_TOP со столбцами EMP_ID, FNAME, LNAME, DEP_ID и SALARY, скопировать в эту таблицу соответствующие данные о сотрудниках, каждый из которых имеет самую высокую зарплату в своём департаменте.

CREATE TABLE EMP_TOP AS
SELECT e.EMPLOYEE_ID AS EMP_ID,
       e.FIRST_NAME AS FNAME,
       e.LAST_NAME AS LNAME,
       e.DEPARTMENT_ID AS DEP_ID,
       e.SALARY
FROM EMPLOYEES e
WHERE e.SALARY = (
    SELECT MAX(inner_e.SALARY)
    FROM EMPLOYEES inner_e
    WHERE inner_e.DEPARTMENT_ID = e.DEPARTMENT_ID -- корреляция
);

-- Проверка:
SELECT * FROM EMP_TOP;



36.	Сотрудникам департамента 30, у которых месячная зарплата ниже половины средней зарплаты по компании, назначить комиссионную надбавку (COMMISSION_PCT) в размере 0.25.

UPDATE EMPLOYEES -- 5 rows updated.
SET COMMISSION_PCT = 0.25
WHERE DEPARTMENT_ID = 30
  AND SALARY < (
    SELECT AVG(SALARY) / 2
    FROM EMPLOYEES
  ); -- 3237.574074074074

-- Проверка:
SELECT EMPLOYEE_ID, SALARY, COMMISSION_PCT
FROM EMPLOYEES
WHERE DEPARTMENT_ID = 30;



40.	Руководителям департаментов, у которых код местоположения (LOCATION_ID) отличается от 1500, 1700 и 2500, установить комиссионную надбавку (COMMISSION_PCT) в раз-мере 0.35.

UPDATE EMPLOYEES
SET COMMISSION_PCT = 0.35
WHERE EMPLOYEE_ID IN ( -- (201, 203, 103, 204), нет корреляции
    SELECT MANAGER_ID
    FROM DEPARTMENTS
    WHERE LOCATION_ID NOT IN (1500, 1700, 2500)
);

-- Проверка:
SELECT EMPLOYEE_ID, COMMISSION_PCT
FROM EMPLOYEES
WHERE EMPLOYEE_ID IN (
    SELECT MANAGER_ID
    FROM DEPARTMENTS
    WHERE LOCATION_ID NOT IN (1500, 1700, 2500)
);
